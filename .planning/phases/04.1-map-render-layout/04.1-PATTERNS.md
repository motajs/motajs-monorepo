  viewport: IMapViewport, metrics: IMapCellMetrics,

  const x = Math.floor((clientX - rect.left - metrics.left) / metrics.size) + viewport.left;

function resolveCellMetrics(document: IMapDocument, model: IMapRenderModel, bigMap: boolean): IMapCellMetrics {
  if (!bigMap) return { size: MAP_CELL_SIZE, left: 0, top: 0 };

async loadRecentTiles(): Promise<void> {
  const token = ++this.recentToken;
  const path = this.resolveRecentTilesPath();

  if (path === undefined) { this.publishRecentTiles(token, []); return; }

  try {
    const resource = await this.context.requestResource(path);
    if (resource === null) { this.publishRecentTiles(token, []); return; }
    const content = await resource.request();
    if (typeof content !== 'string') { this.publishRecentTiles(token, []); return; }
    this.publishRecentTiles(token, this.parseRecentTiles(content, this.collectTileNumbers()));
  } catch {
    this.publishRecentTiles(token, []);
  }
}

async writeRecentTiles(): Promise<void> {
  const path = this.resolveRecentTilesPath();
  if (path === undefined) return;

  const text = serializeTableText({ [MAP_RECENT_TILES_KEY]: this.store.state.recentTiles }, '');

  try {
    const resource = await this.context.requestResource(path);
    if (resource !== null) { await resource.write(text); return; }
    await this.context.createResource(path, text);
  } catch {
    return;
  }
}

- 不用 `?.`（仅副作用调用与 Required 化两种场景）；不用单个对象解构。
- 只有 `createEditorCore` 这类**有状态工厂**才用类；纯功能函数保持裸函数。
